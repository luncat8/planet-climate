Вот итоговый план реализации визуальных улучшений (Streamlines), основанный на ваших вариантах **1 (Temporal Accumulation)**, **2 (Spatial Smoothing)**, **3 (Coherence Seeding)** и **6 (Trail Fade/Length)**.

Эти изменения не затрагивают физику симуляции (`OCEAN_FS`, `AIR_FS`), а работают исключительно в пайплайне частиц и рендеринга.

---

### 📋 План внедрения

1.  **Shader.js**: Модификация `PART_VS` (сглаживание + чтение накопителя), `PART_FS` (когерентность при спавне), `PART_PS` (затухание хвоста).
2.  **Engine.js**: Добавление FBO для temporal accumulation (ping-pong буфер скорости) и нового прохода обновления перед `stepParticles`.
3.  **App.js**: Добавление UI-контролов (чекбоксы/слайдеры) в секцию "Tuning" или отдельную секцию "Visualization".

### 1. Модификация `shader.js`

Добавляем новые uniforms: `uSmoothVel`, `uTemporalAcc`, `uCoherence`, `uTrailFade`.

#### A. `PART_VS` — Сглаживание и чтение накопителя
Заменяем простой fetch скорости на логику с учетом соседей или накопленного буфера.

```glsl
// В PART_VS, перед блоком выборки скорости:
uniform float uSmoothVel;   // > 0.5 = spatial smoothing (Option 2)
uniform sampler2D uVelAcc;  // Temporal accumulation buffer (Option 1)
uniform int   uUseAccum;    // > 0.5 = read from uVelAcc instead of raw velocity

// ... (существующий код выборки cell, e1, e2) ...

vec2 vel;
if (uUseAccum > 0.5) {
    // Читаем из накопителя (Option 1)
    vel = texelFetch(uVelAcc, cTex(cell), 0).xy;
} else if (uSmoothVel > 0.5) {
    // Spatial Smoothing (Option 2): усреднение по 1-ring
    vec2 sumV = vec2(0.0);
    float wSum = 0.0;
    for(int k=0; k<6; k++){
        vec4 na = texelFetch(uNbrA, nTex(cell,k), 0);
        if(na.w < 0.5) continue;
        int j = int(na.x);
        vec2 vj = xfer(texelFetch(uTopV, cTex(j), 0).xy, nb.z, nb.w); // Нужно подтянуть nb из контекста или пересчитать
        // Упрощенно: берем вектор соседа, трансформируем в локальную систему текущего cell
        vec2 vjLocal = xfer(vj, nb.z, nb.w); 
        sumV += vjLocal;
        wSum += 1.0;
    }
    if(wSum > 0.0) vel = sumV / wSum;
    else vel = texelFetch(uTopV, cTex(cell), 0).xy;
} else {
    // Legacy: raw velocity
    if(uVelMode==0)      vel = texelFetch(uLoA, cTex(cell),0).xy;
    else if(uVelMode==1) vel = texelFetch(uTopV, cTex(cell),0).xy*uVelScale;
    else if(uVelMode==2) vel = texelFetch(uHiA, cTex(cell),0).xy;
    else                 vel = texelFetch(uDeepV, cTex(cell),0).xy*uVelScale;
}

vec3 v3 = vel.x*e1 + vel.y*e2;
```

#### B. `PART_FS` — Когерентность при спавне (Option 3)
В блоке респауна (`age <= 0`) добавляем проверку на "гладкость" потока. Если поле зашумлено, частица не возрождается (или возрождается с меньшей вероятностью).

```glsl
// В PART_FS, в блоке if(age <= 0.0):
if(uCoherence > 0.5) {
    // Проверка когерентности: curl и divergence через соседей
    vec2 sumCurl = vec2(0.0);
    float sumDiv = 0.0;
    for(int k=0; k<6; k++){
        vec4 na = texelFetch(uNbrA, nTex(cell,k), 0);
        if(na.w < 0.5) continue;
        int j = int(na.x);
        float d = na.z;
        vec2 vj = xfer(texelFetch(uTopV, cTex(j), 0).xy, nb.z, nb.w) * uVelScale; // Нужно адаптировать под mode
        // Curl approx: (dv/dx - du/dy)
        sumCurl += (vj.y - vt0.y) / d; 
        sumDiv  += dot(vj - vt0, vec2(1.0)) / d; // Грубая аппроксимация дивергенции
    }
    float curlMag = length(sumCurl);
    float divMag  = abs(sumDiv);
    
    // Если поток неупорядоченный (низкий curl или высокий div), пропускаем спавн
    if(curlMag < 0.1 || divMag > 0.5) {
        pos = vec3(0.0, -1.0, 0.0); // "Мертвая" частица
        return; 
    }
}

// ... (существующий randDir и нормализация) ...
```

#### C. `PART_PS` — Затухание хвоста (Option 6)
Добавляем fade по возрасту, чтобы старые сегменты плавно исчезали.

```glsl
void main(){
    float a = vA * 0.75;
    
    // Fade by age: чем старше частица (p.w ближе к 1), тем прозрачнее хвост
    if(uTrailFade > 0.5) {
        float ageFactor = clamp(1.5 - abs(p.w-0.5)*2.0, 0.0, 1.0); // Упрощенная формула из PART_VS
        a *= ageFactor;
    }

    if (uAsPoints > 0.5) {
        float d = length(gl_PointCoord - 0.5);
        a *= smoothstep(0.5, 0.15, d);
    }
    o = vec4(uColor, a);
}
```

### 2. Модификация `engine.js`

Добавляем FBO для temporal accumulation и проход обновления.

#### A. Инициализация текстур (в `Planet.prototype.build`)
```javascript
// Текстуры для накопления скорости (ping-pong)
this.texVelAccA = this.mkTex(this.PW, this.PH, null);
this.texVelAccB = this.mkTex(this.PW, this.PH, null);
this.fbo.velAccA = this.mkFbo([this.texVelAccA]);
this.fbo.velAccB = this.mkFbo([this.texVelAccB]);
```

#### B. Новый шейдер для накопления (добавить в `shader.js`)
```glsl
var VEL_ACC_FS = SHADER_HEAD + `
layout(location=0) out vec4 o;
uniform sampler2D uSrc, uPrev; // uSrc = current vel, uPrev = prev frame acc
uniform float uAccAlpha;       // blending factor (e.g., 0.15)
void main(){
    vec2 cur = texelFetch(uSrc, ivec2(gl_FragCoord.xy), 0).xy;
    vec2 prev = texelFetch(uPrev, ivec2(gl_FragCoord.xy), 0).xy;
    o = vec4(mix(prev, cur, uAccAlpha), 0.0, 1.0);
}`;
```

#### C. Логика шага (в `Planet.prototype.stepParticles` или новый метод)
```javascript
Planet.prototype.updateVelAccumulation = function() {
    var P = this.params;
    if(P.visTemporalAcc <= 0.5) return; // Опционально
    
    var gl = this.gl;
    var src = 'velAccA', dst = 'velAccB';
    
    var p = this.prog.velAcc.use();
    this.gridUniforms(p);
    // Читаем текущую скорость из A[1] (topV)
    p.tex('uSrc', this.A[1]).tex('uPrev', this[src])
      .f('uAccAlpha', P.visAccAlpha || 0.2);
    
    this.fullscreen(dst, this.PW, this.PH);
    
    // Swap ping-pong
    var t = src; src = dst; dst = t;
    this.texVelAccA = this[src] === this.texVelAccA ? this.texVelAccA : this.texVelAccB; 
    // (Упрощенно: просто меняем ссылки в фреймбуфферах)
};

// Вызывать перед stepParticles:
Planet.prototype.stepParticles = function (dt) {
  this.updateVelAccumulation();
  // ... existing code ...
  // Передаем uVelAcc и uUseAccum в PART_VS через uniform
  p.tex('uVelAcc', this.texVelAccA); 
};
```

### 3. Модификация `app.js` (UI)

Добавляем контролы в секцию тюнинга.

```javascript
// В buildUI(), после блока чекбоксов:
panel.appendChild(el('div', 'lab', 'Streamline Visualization'));

var visDefs = [
    ['visSmoothVel', 'Spatial Smoothing (Neighbors)'], // Option 2
    ['visTemporalAcc', 'Temporal Accumulation (History)'], // Option 1
    ['visCoherence', 'Coherence Seeding'], // Option 3
    ['visTrailFade', 'Trail Fade (Age-based)'], // Option 6
];

visDefs.forEach(function(d) {
    var lab = el('label', 'chk');
    lab.appendChild(el('span', null, d[1]));
    var inp = document.createElement('input');
    inp.type = 'checkbox';
    inp.className = 'accent-cyan-400';
    inp.onchange = function() { setParam(d[0], inp.checked ? 1 : 0); };
    ui.checks[d[0]] = inp;
    lab.appendChild(inp);
    panel.appendChild(lab);
});

// Для Temporal Accumulation можно добавить слайдер скорости затухания:
var accWrap = el('div');
accWrap.style.marginBottom = '10px';
accWrap.appendChild(el('span', null, 'Accumulation Rate'));
var accInp = document.createElement('input');
accInp.type = 'range';
accInp.min = 0.05; accInp.max = 0.5; accInp.step = 0.05; accInp.value = 0.2;
accInp.oninput = function() { setParam('visAccAlpha', parseFloat(accInp.value)); };
accWrap.appendChild(accInp);
panel.appendChild(accWrap);
```

### 📊 Сводная таблица изменений

| Фича | Шейдер | Engine | UI | Эффект |
|------|--------|--------|----|--------|
| **Spatial Smoothing** (2) | `PART_VS` (loop neighbors) | - | Чекбокс `visSmoothVel` | Убирает мелкий шум в линиях |
| **Temporal Accumulation** (1) | Новый FBO pass + `uVelAcc` | FBO ping-pong | Чекбокс `visTemporalAcc` + слайдер | Сглаживает движение во времени |
| **Coherence Seeding** (3) | `PART_FS` (curl/div check) | - | Чекбокс `visCoherence` | Частицы только в течениях, не в шуме |
| **Trail Fade** (6) | `PART_PS` (alpha fade) | - | Чекбокс `visTrailFade` | Красивое затухание хвостов |

Этот план позволяет пользователю комбинировать эффекты: например, включить **Smoothing + Accumulation** для максимального сглаживания или только **Coherence** для чистых линий без изменения динамики.
